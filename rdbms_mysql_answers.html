MySQL Schema (CREATE TABLE)

  client_no  INT REFERENCES Client(client_no),

FUNCTIONReturn total maturity amount for a particular client
Approach: Accept client_no. JOIN Client_Policy with Policy_info and SUM maturity_amt. Return the total using IFNULL.
MySQL Tags:DROP FUNCTION IF EXISTSCREATE FUNCTIONRETURNS DECIMALSELECT INTOJOINSUMIFNULLDETERMINISTIC
💡 MySQL functions need DETERMINISTIC or READS SQL DATA keyword.
DELIMITER $$

DROP FUNCTION IF EXISTS get_total_maturity$$
CREATE FUNCTION get_total_maturity(p_client INT)
RETURNS DECIMAL(10,2)
READS SQL DATA
BEGIN
  DECLARE v_total DECIMAL(10,2) DEFAULT 0;

  SELECT IFNULL(SUM(p.maturity_amt), 0)
    INTO v_total
    FROM Policy_info p
    JOIN Client_Policy cp ON p.policy_no = cp.policy_no
   WHERE cp.client_no = p_client;

  RETURN v_total;
END$$

CURSORDisplay policy date-wise client details
Approach: Use a cursor inside a stored procedure. Loop through JOIN of Client, Client_Policy, Policy_info ordered by policy_date. Use SELECT to display each row.
MySQL Tags:DELIMITER $$CREATE PROCEDUREDECLARE CURSORDECLARE CONTINUE HANDLEROPEN/FETCH/CLOSECONCATSELECT output
💡 MySQL has no DBMS_OUTPUT. Use SELECT CONCAT(...) AS Output inside procedures to print results.
DELIMITER $$

DROP PROCEDURE IF EXISTS policy_datewise$$
CREATE PROCEDURE policy_datewise()
BEGIN
  DECLARE done INT DEFAULT 0;
  DECLARE v_cname  VARCHAR(50);
  DECLARE v_pno    INT;
  DECLARE v_pdesc  VARCHAR(100);
  DECLARE v_prem   DECIMAL(10,2);
  DECLARE v_mat    DECIMAL(10,2);
  DECLARE v_pdate  DATE;

  DECLARE cur CURSOR FOR
    SELECT c.client_name, p.policy_no, p.policy_desc,
           p.prem_amt, p.maturity_amt, p.policy_date
      FROM Client c
      JOIN Client_Policy cp ON c.client_no = cp.client_no
      JOIN Policy_info p    ON p.policy_no  = cp.policy_no
     ORDER BY p.policy_date;

  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

  OPEN cur;
  read_loop: LOOP
    FETCH cur INTO v_cname,v_pno,v_pdesc,v_prem,v_mat,v_pdate;
    IF done THEN LEAVE read_loop; END IF;
    SELECT CONCAT(
      'Date: '      , v_pdate,
      ' | Client: ' , v_cname,
      ' | PolicyNo: ', v_pno,
      ' | Prem: '   , v_prem,
      ' | Maturity: ', v_mat) AS Output;
  END LOOP;
  CLOSE cur;
END$$

DELIMITER ;

PROCEDURECalculate city-wise total cost of each newspaper
Approach: GROUP BY city, compute SUM(cost * daily_required). Use SELECT to display results directly from the procedure.
MySQL Tags:CREATE PROCEDURESELECT with GROUP BYJOINSUMORDER BY
💡 In MySQL procedures, a plain SELECT produces a result set visible in Workbench output panel.
DELIMITER $$

DROP PROCEDURE IF EXISTS city_newspaper_cost$$
CREATE PROCEDURE city_newspaper_cost()
BEGIN
  SELECT ci.city,
         SUM(n.cost * nc.daily_required) AS Total_Cost
    FROM Cities ci
    JOIN Newspaper_City nc ON ci.pincode  = nc.pincode
    JOIN Newspaper n       ON n.name      = nc.name
   GROUP BY ci.city
   ORDER BY ci.city;
END$$

CURSORDisplay library-wise book details (Parameterized Cursor equivalent)
Approach: MySQL has no parameterized cursors. Use an outer loop over Library rows, then a SELECT inside the loop filtered by current Lno.
MySQL Tags:CREATE PROCEDUREDECLARE CURSORDECLARE CONTINUE HANDLER FOR NOT FOUNDOPEN/FETCH/CLOSENested SELECTCONCAT
💡 MySQL parameterized cursor workaround: use a regular SELECT inside the loop with the loop variable as filter.
DELIMITER $$

DROP PROCEDURE IF EXISTS lib_book_details$$
CREATE PROCEDURE lib_book_details()
BEGIN
  DECLARE done    INT DEFAULT 0;
  DECLARE v_lno   INT;
  DECLARE v_lname VARCHAR(50);

  DECLARE cur_lib CURSOR FOR
    SELECT Lno, Lname FROM Library;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

  OPEN cur_lib;
  lib_loop: LOOP
    FETCH cur_lib INTO v_lno, v_lname;
    IF done THEN LEAVE lib_loop; END IF;

    SELECT CONCAT('=== Library: ' , v_lname , ' ===' ) AS Header;

    -- Nested SELECT for books of this library
    SELECT Bname, Author_Name, Price, publication
      FROM Book
     WHERE Lno = v_lno;

  END LOOP;
  CLOSE cur_lib;
END$$

CURSORDisplay customer-wise plan details
Approach: Outer cursor loops over Plan. For each plan, nested SELECT fetches customers with that plan_no.
MySQL Tags:CREATE PROCEDURECURSORDECLARE CONTINUE HANDLEROPEN/FETCH/CLOSENested SELECT

CREATE PROCEDURE customer_by_plan()
BEGIN
  DECLARE done INT DEFAULT 0;
  DECLARE v_pno   INT;
  DECLARE v_pname VARCHAR(50);
  DECLARE v_amt   DECIMAL(8,2);

  DECLARE cur_plan CURSOR FOR
    SELECT plan_no, plan_name, fix_amt FROM Plan;

    SELECT cust_no, cust_name, mobile_no
      FROM Customer
     WHERE plan_no = v_pno;
  END LOOP;
  CLOSE cur_plan;
END$$

CREATE FUNCTION progressive_count(p_dept VARCHAR(50))

READS SQL DATA
BEGIN
  DECLARE v_cnt INT DEFAULT 0;
  SELECT COUNT(*) INTO v_cnt
    FROM Project p

CREATE PROCEDURE club_player_details()
BEGIN
  DECLARE done   INT         DEFAULT 0;
  DECLARE v_prev VARCHAR(50) DEFAULT '~';
  DECLARE v_pid   INT;
  DECLARE v_pname VARCHAR(50);
  DECLARE v_club  VARCHAR(50);

  DECLARE cur CURSOR FOR
    SELECT pid, pname, club_name FROM Player ORDER BY club_name;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=1;

  OPEN cur;
  lp: LOOP
    FETCH cur INTO v_pid,v_pname,v_club;
    IF done THEN LEAVE lp; END IF;
    IF v_club != v_prev THEN
      SELECT CONCAT('=== Club: ',v_club,' ===' ) AS Club_Header;
      SET v_prev = v_club;
    END IF;
    SELECT CONCAT('PID:' ,v_pid,' Name:' ,v_pname) AS Player_Row;
  END LOOP;
  CLOSE cur;
END$$

CREATE PROCEDURE cars_on_day(IN p_date DATE)
BEGIN
  SELECT DISTINCT c.license_no, c.model, c.year, dc.usage_time
    FROM Car c
    JOIN Driver_Car dc ON c.license_no = dc.license_no
   WHERE DATE(dc.usage_date) = p_date;
END$$

CREATE PROCEDURE driver_cars_2018()
BEGIN
  DECLARE done    INT DEFAULT 0;
  DECLARE v_prev  INT DEFAULT -1;
  DECLARE v_did   INT;
  DECLARE v_dname VARCHAR(50);
  DECLARE v_lic   VARCHAR(20);
  DECLARE v_model VARCHAR(50);
  DECLARE v_udate DATE;
  DECLARE v_utime VARCHAR(10);

  DECLARE cur CURSOR FOR
    SELECT d.driver_id, d.driver_name,
           c.license_no, c.model, dc.usage_date, dc.usage_time
      FROM Driver d
      JOIN Driver_Car dc ON d.driver_id  = dc.driver_id
      JOIN Car c         ON c.license_no = dc.license_no
     WHERE YEAR(dc.usage_date) = 2018
     ORDER BY d.driver_id;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=1;

  OPEN cur;
  lp: LOOP
    FETCH cur INTO v_did,v_dname,v_lic,v_model,v_udate,v_utime;
    IF done THEN LEAVE lp; END IF;
    IF v_did != v_prev THEN
      SELECT CONCAT('=== Driver: ',v_dname,' ===' ) AS Driver_Header;
      SET v_prev = v_did;
    END IF;
    SELECT CONCAT(
      'Car: ' ,v_model,' | Lic: ' ,v_lic,
      ' | Date: ' ,v_udate,' | Time: ' ,v_utime) AS Car_Row;
  END LOOP;
  CLOSE cur;
END$$

DELIMITER ;
