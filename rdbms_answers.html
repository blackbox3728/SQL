📌 PL/SQL Key Tags & Definitions

CREATE TABLE
DDL to define a table with columns, data types, and constraints (PRIMARY KEY, NOT NULL, CHECK, FOREIGN KEY).
CREATE OR REPLACE FUNCTION
Named PL/SQL block that returns a single value. Must contain a RETURN clause. Called inside SQL or PL/SQL.
CREATE OR REPLACE PROCEDURE
Named PL/SQL block that performs actions. Does NOT return a value (uses IN/OUT params). Invoked with EXEC.
CREATE OR REPLACE TRIGGER
Auto-executed block that fires on DML events (INSERT/UPDATE/DELETE) BEFORE or AFTER the operation.
CURSOR
Pointer to a result set. Types: Implicit (auto), Explicit (DECLARE/OPEN/FETCH/CLOSE), Parameterized (accepts parameters).
DECLARE … BEGIN … END
Sections of a PL/SQL anonymous block. DECLARE: variables/cursors. BEGIN: executable code. END: closes the block.
EXCEPTION
Error-handling section. User-defined exceptions declared in DECLARE and raised with RAISE keyword.
RAISE_APPLICATION_ERROR
Raises a custom error. Syntax: RAISE_APPLICATION_ERROR(-20001,'msg'). Error numbers -20001 to -20999.
FOR rec IN cursor LOOP
Implicit FOR loop — auto opens, fetches, and closes the cursor. Simplest way to iterate a result set.
OPEN / FETCH / CLOSE
Explicit cursor control. OPEN executes query; FETCH retrieves next row; CLOSE releases cursor resources.
%ROWTYPE / %TYPE
%ROWTYPE inherits all columns of a table. %TYPE inherits the data type of a single column. Ensures compatibility.
:NEW / :OLD
Trigger pseudo-records. :NEW = value being inserted/updated. :OLD = value before update/delete.
BEFORE INSERT OR UPDATE
Trigger timing — fires BEFORE each INSERT or UPDATE DML statement on the target table.
FOR EACH ROW
Row-level trigger — fires once per affected row. Required to access :NEW and :OLD pseudo-records.
DBMS_OUTPUT.PUT_LINE
Built-in procedure to print text output. Enable with SET SERVEROUTPUT ON in SQL*Plus before running.
SELECT INTO
PL/SQL SELECT that stores the result into a variable. Raises NO_DATA_FOUND if no row matches; TOO_MANY_ROWS if multiple.

  policy_no  NUMBER  REFERENCES Policy_info(policy_no),
  PRIMARY KEY(client_no, policy_no)

FUNCTIONReturn total maturity amount of policies for a particular client
Approach: Accept client_no as IN parameter. JOIN Client_Policy with Policy_info and SUM maturity_amt for that client. Use NVL to handle NULL.
Tags used:CREATE FUNCTIONRETURN NUMBERSELECT INTOJOINSUMNVL
CREATE OR REPLACE FUNCTION get_total_maturity(p_client IN Client.client_no%TYPE)
RETURN NUMBER IS
  v_total NUMBER := 0;
BEGIN
  SELECT SUM(p.maturity_amt)
    INTO v_total
    FROM Policy_info p
    JOIN Client_Policy cp ON p.policy_no = cp.policy_no
   WHERE cp.client_no = p_client;
  RETURN NVL(v_total, 0);
END;
/
-- Calling the function:

BEGIN
  v := get_total_maturity(101);
  DBMS_OUTPUT.PUT_LINE('Total Maturity Amount: ' || v);
END;
/

PROCEDURECalculate city-wise total cost of each newspaper
Approach: Cursor JOINs Cities, Newspaper_City, and Newspaper. GROUP BY city, compute SUM(cost × daily_required). Print results.
Tags used:PROCEDURECURSORFOR LOOPGROUP BYSUMRPAD
CREATE OR REPLACE PROCEDURE city_newspaper_cost IS
  CURSOR c_cc IS
    SELECT ci.city,
           SUM(n.cost * nc.daily_required) AS total_cost
      FROM Cities ci
      JOIN Newspaper_City nc ON ci.pincode = nc.pincode
      JOIN Newspaper n       ON n.name     = nc.name
     GROUP BY ci.city
     ORDER BY ci.city;
BEGIN
  DBMS_OUTPUT.PUT_LINE(RPAD('City',20) || 'Total Cost');
  DBMS_OUTPUT.PUT_LINE(RPAD('-',30,'-'));
  FOR r IN c_cc LOOP
    DBMS_OUTPUT.PUT_LINE(RPAD(r.city,20) || r.total_cost);
  END LOOP;
END;
/

CURSORDisplay library-wise book details using Parameterized Cursor
Approach: Outer loop fetches each library. For each Lno, a parameterized cursor is opened to fetch its books. OPEN/FETCH/CLOSE pattern used.
Tags used:PARAMETERIZED CURSOROPEN cursor(param)FETCHCLOSE%ROWTYPENested LOOP
DECLARE
  CURSOR c_lib IS
    SELECT Lno, Lname FROM Library;

  -- Parameterized cursor accepts Lno
  CURSOR c_books(p_lno Library.Lno%TYPE) IS
    SELECT Bid, Bname, Author_Name, Price, publication
      FROM Book WHERE Lno = p_lno;

  v_bk c_books%ROWTYPE;
BEGIN
  FOR lib IN c_lib LOOP
    DBMS_OUTPUT.PUT_LINE('=== Library: ' || lib.Lname || ' ===');
    OPEN c_books(lib.Lno);
    LOOP
      FETCH c_books INTO v_bk;
      EXIT WHEN c_books%NOTFOUND;
      DBMS_OUTPUT.PUT_LINE(
        '  Book: '    || v_bk.Bname       ||
        ' | Author: ' || v_bk.Author_Name ||
        ' | Price: '  || v_bk.Price       ||
        ' | Pub: '    || v_bk.publication);
    END LOOP;
    CLOSE c_books;
  END LOOP;
END;
/

PROCEDUREDisplay menu items priced 200–500 ordered on Saturday
Approach: JOIN Bill_Menu with Menu and Bill. Filter WHERE price BETWEEN 200 AND 500 AND UPPER(day) = 'SATURDAY'. Display via cursor loop.

CURSORDisplay customer-wise plan details using Parameterized Cursor
Approach: Outer loop iterates each Plan. For each plan_no, parameterized cursor fetches subscribed customers. OPEN/FETCH/CLOSE used explicitly.
Tags used:PARAMETERIZED CURSOROPEN/FETCH/CLOSE%ROWTYPEFOR LOOPNested LOOP
DECLARE
  CURSOR c_plan IS
    SELECT plan_no, plan_name, fix_amt FROM Plan;

  CURSOR c_cust(p_plan Plan.plan_no%TYPE) IS
    SELECT cust_no, cust_name, mobile_no
      FROM Customer WHERE plan_no = p_plan;

  v_c c_cust%ROWTYPE;
BEGIN
  FOR p IN c_plan LOOP
    DBMS_OUTPUT.PUT_LINE(

      DBMS_OUTPUT.PUT_LINE(
        '  ' || v_c.cust_name || ' (' || v_c.mobile_no || ')');
    END LOOP;
    CLOSE c_cust;
  END LOOP;
END;

-- Status: C=Completed, P=Progressive, I=Incomplete
FUNCTIONAccept department name and return count of Progressive (P) projects
Approach: JOIN Project with Department. COUNT(*) WHERE status = 'P' AND UPPER(dname) = UPPER(parameter). Return count.
Tags used:FUNCTIONJOINCOUNT(*)SELECT INTOUPPER()
CREATE OR REPLACE FUNCTION progressive_count(p_dept IN Department.dname%TYPE)
RETURN NUMBER IS
  v_cnt NUMBER;
BEGIN
  SELECT COUNT(*) INTO v_cnt
    FROM Project p
    JOIN Department d ON p.dno = d.dno
   WHERE p.status = 'P'
     AND UPPER(d.dname) = UPPER(p_dept);
  RETURN v_cnt;
END;
/
BEGIN
  DBMS_OUTPUT.PUT_LINE('Progressive Projects: ' || progressive_count('IT Department'));
END;

CURSORDisplay status-wise project details of each department
Approach: Cursor JOINs Department and Project ordered by dname, status. Track department changes with a variable to print group headers.
Tags used:CURSORFOR LOOPJOINORDER BYChange-break logic
DECLARE
  CURSOR c_sp IS
    SELECT d.dname, p.pno, p.pname,
           p.status, p.budget
      FROM Department d
      JOIN Project p ON d.dno = p.dno
     ORDER BY d.dname, p.status;
  v_prev Department.dname%TYPE := '~';
BEGIN
  FOR r IN c_sp LOOP
